1"""Reject HDF5/NetCDF files that reference storage outside themselves. 

2 

3HDF5 external storage, external links and virtual datasets make a variable's 

4bytes resolve from a separate, file-embedded path only when the data is *read*. 

5Inspecting metadata does not dereference them, so untrusted files can be 

6screened and rejected before anything is opened through xarray/h5py. 

7""" 

8 

9from typing import List, Set 

10 

11import h5py 

12import h5py.h5o as h5o 

13 

14 

15class ExternalReferenceError(Exception): 

16 """Raised when a file points at storage outside itself.""" 

17 

18 

19def assert_no_external_references(paths: List[str]) -> None: 

20 """Raise ExternalReferenceError if any file uses external storage, an 

21 external link or a virtual dataset. Only metadata is read, never data.""" 

22 for path in paths: 

23 with h5py.File(path, "r") as file: 

24 findings: List[str] = [] 

25 _scan_group(file, findings, set()) 

26 if findings: 

27 raise ExternalReferenceError( 

28 f"Refusing to open {path}: it references storage outside the file " 

29 f"({'; '.join(findings)}). External storage, external links and " 

30 "virtual datasets are not permitted." 

31 ) 

32 

33 

34def _scan_group(group: h5py.Group, findings: List[str], visited: Set[int]) -> None: 

35 for key in group.keys(): 

36 # Inspect the link itself first so external links are never dereferenced. 

37 link = group.get(key, getlink=True) 

38 if isinstance(link, h5py.ExternalLink): 

39 parent = (group.name or "").rstrip("/") 

40 findings.append(f"external link '{parent}/{key}' -> {link.filename}") 

41 continue 

42 if isinstance(link, h5py.SoftLink): 

43 # Stays within the file; its hard-linked target is scanned on its own. 

44 continue 

45 

46 item = group[key] 

47 addr = h5o.get_info(item.id).addr 

48 if addr in visited: # guard against cyclic hard links 

49 continue 

50 visited.add(addr) 

51 

52 if isinstance(item, h5py.Group): 

53 _scan_group(item, findings, visited) 

54 elif isinstance(item, h5py.Dataset): 

55 _scan_dataset(item, findings) 

56 

57 

58def _scan_dataset(dataset: h5py.Dataset, findings: List[str]) -> None: 

59 if dataset.is_virtual: 

60 findings.append(f"virtual dataset '{dataset.name}'") 

61 if dataset.id.get_create_plist().get_external_count() > 0: 

62 findings.append(f"external storage '{dataset.name}'")
